813{

816 /* For Suite B mode only include P-256, P-384 */
817 switch (tls1_suiteb(s)(s->cert->cert_flags & 0x30000)) {

849 * eligible group from the legacy list. Therefore, we provide the entire list of supported

1025/*-
1026 * For nmatch >= 0, return the id of the |nmatch|th shared group or 0
1027 * if there is no match.
1028 * For nmatch == TLS1_GROUPS_RETURN_NUMBER, return number of matches
1029 * For nmatch == TLS1_GROUPS_RETURN_TMP_ID, return the id of the group to use
1030 * for a tmp key, or 0 if there is no match.
1031 * If groups == TLS1_GROUPS_FFDHE_GROUPS, only shared groups that are FFDHE
1032 * groups (i.e., between OSSL_TLS_GROUP_ID_FFDHE_START and
1033 * OSSL_TLS_GROUP_ID_FFDHE_END, inclusive) will be included in the search.
1034 * If groups == TLS1_GROUPS_NON_FFDHE_GROUPS, only shared groups that are not
1035 * FFDHE groups will be included in the search.
1036 * If groups == TLS1_GROUPS_ALL_GROUPS, all groups will be included in the
1037 * search.

1040{

1042 size_t num_pref, num_supp, i;
1043 int k;
1044 SSL_CTX *ctx = SSL_CONNECTION_GET_CTX(s)((s)->ssl.ctx);
1045
1046 /* Can't do anything on client side */
1047 if (s->server == 0)
1048 return 0;

1050 if (groups != TLS1_GROUPS_FFDHE_GROUPS0 && tls1_suiteb(s)(s->cert->cert_flags & 0x30000)) {
1051 /*
1052 * For Suite B ciphersuite determines curve: we already know
1053 * these are acceptable due to previous checks.
1054 */
1055 unsigned long cid = s->s3.tmp.new_cipher->id;
1056
1057 if (cid == TLS1_CK_ECDHE_ECDSA_WITH_AES_128_GCM_SHA2560x0300C02B)
1058 return OSSL_TLS_GROUP_ID_secp256r10x0017;
1059 if (cid == TLS1_CK_ECDHE_ECDSA_WITH_AES_256_GCM_SHA3840x0300C02C)
1060 return OSSL_TLS_GROUP_ID_secp384r10x0018;
1061 /* Should never happen */
1062 return 0;
1063 }
1064 /* If not Suite B just return first preference shared curve */
1065 nmatch = 0;
1066 }

1068 * If server preference set, our groups are the preference order
1069 * otherwise peer decides.
1070 */
1071 if (s->options & SSL_OP_SERVER_PREFERENCE((uint64_t)1 << (uint64_t)22)) {
1072 tls1_get_supported_groups(s, &pref, &num_pref);
1073 tls1_get_peer_groups(s, &supp, &num_supp);
1074 } else {
1075 tls1_get_peer_groups(s, &pref, &num_pref);
1076 tls1_get_supported_groups(s, &supp, &num_supp);
1077 }
1078
1079 for (k = 0, i = 0; i < num_pref; i++) {
1080 uint16_t id = pref[i];
1081 const TLS_GROUP_INFO *inf;
1082 int minversion, maxversion;
1083
1084 if (!tls1_in_list(id, supp, num_supp)
1085 || (groups == TLS1_GROUPS_NON_FFDHE_GROUPS1 && is_ffdhe_group(id))
1086 || (groups == TLS1_GROUPS_FFDHE_GROUPS0 && !is_ffdhe_group(id))
1087 || !tls_group_allowed(s, id, SSL_SECOP_CURVE_SHARED(5 | (2 << 16))))
1088 continue;

1090 if (!ossl_assert(inf != NULL)__builtin_expect(!!((inf != ((void*)0)) != 0), 1))
1091 return 0;

0x8)
1094 ? inf->mindtls
1095 : inf->mintls;
1096 maxversion = SSL_CONNECTION_IS_DTLS(s)((&(s)->ssl)->method->ssl3_enc->enc_flags &
0x8)
1097 ? inf->maxdtls
1098 : inf->maxtls;
1099 if (maxversion == -1)
1100 continue;
1101 if ((minversion != 0 && ssl_version_cmp(s, s->version, minversion) < 0)
1102 || (maxversion != 0
1103 && ssl_version_cmp(s, s->version, maxversion) > 0))
1104 continue;
1105
1106 if (nmatch == k)
1107 return id;
1108 k++;
1109 }
1110 if (nmatch == TLS1_GROUPS_RETURN_NUMBER-1)
1111 return k;
1112 /* Out of range (nmatch > k). */
1113 return 0;
1114}

1866/* Check a group id matches preferences */
1867int tls1_check_group_id(SSL_CONNECTION *s, uint16_t group_id,
1868 int check_own_groups)
1869{
1870 const uint16_t *groups;
1871 size_t groups_len;
1872
1873 if (group_id == 0)
1874 return 0;
1875
1876 /* Check for Suite B compliance */
1877 if (tls1_suiteb(s)(s->cert->cert_flags & 0x30000) && s->s3.tmp.new_cipher != NULL((void*)0)) {
1878 unsigned long cid = s->s3.tmp.new_cipher->id;
1879
1880 if (cid == TLS1_CK_ECDHE_ECDSA_WITH_AES_128_GCM_SHA2560x0300C02B) {
1881 if (group_id != OSSL_TLS_GROUP_ID_secp256r10x0017)
1882 return 0;
1883 } else if (cid == TLS1_CK_ECDHE_ECDSA_WITH_AES_256_GCM_SHA3840x0300C02C) {
1884 if (group_id != OSSL_TLS_GROUP_ID_secp384r10x0018)
1885 return 0;
1886 } else {
1887 /* Should never happen */
1888 return 0;
1889 }
1890 }
1891
1892 if (check_own_groups) {
1893 /* Check group is one of our preferences */
1894 tls1_get_supported_groups(s, &groups, &groups_len);
1895 if (!tls1_in_list(group_id, groups, groups_len))
1896 return 0;
1897 }
1898
1899 if (!tls_group_allowed(s, group_id, SSL_SECOP_CURVE_CHECK(6 | (2 << 16))))
1900 return 0;
1901
1902 /* For clients, nothing more to check */
1903 if (!s->server)
1904 return 1;
1905
1906 /* Check group is one of peers preferences */
1907 tls1_get_peer_groups(s, &groups, &groups_len);
1908
1909 /*
1910 * RFC 4492 does not require the supported elliptic curves extension
1911 * so if it is not sent we can just choose any curve.
1912 * It is invalid to send an empty list in the supported groups
1913 * extension, so groups_len == 0 always means no extension.
1914 */
1915 if (groups_len == 0)
1916 return 1;
1917 return tls1_in_list(group_id, groups, groups_len);
1918}

1991/*
1992 * tls1_check_ffdhe_tmp_key - Check FFDHE temporary key compatibility
1993 * @s: SSL connection
1994 * @cid: Cipher ID we're considering using
1995 *
1996 * Checks that the kDHE cipher suite we're considering using
1997 * is compatible with the client extensions.
1998 *
1999 * Returns 0 when the cipher can't be used or 1 when it can.
2000 */
2001int tls1_check_ffdhe_tmp_key(SSL_CONNECTION *s, unsigned long cid)
2002{
2003 const uint16_t *peer_groups;
2004 size_t num_peer_groups;
2005
2006 /* If we have a shared FFDHE group, we can certainly use it. */
2007 if (tls1_shared_group(s, 0, TLS1_GROUPS_FFDHE_GROUPS0) != 0)
2008 return 1;
2009
2010 /*
2011 * Otherwise, we follow RFC 7919:
2012 * If a compatible TLS server receives a Supported Groups extension from
2013 * a client that includes any FFDHE group (i.e., any codepoint between
2014 * 256 and 511, inclusive, even if unknown to the server), and if none
2015 * of the client-proposed FFDHE groups are known and acceptable to the
2016 * server, then the server MUST NOT select an FFDHE cipher suite.
2017 */
2018 tls1_get_peer_groups(s, &peer_groups, &num_peer_groups);
2019 for (size_t i = 0; i < num_peer_groups; i++) {
2020 if (is_ffdhe_group(peer_groups[i]))
2021 return 0;
2022 }
2023
2024 /*
2025 * The client did not send any FFDHE groups, so we can use this ciphersuite
2026 * using any group we like.
2027 */
2028 return 1;
2029}

2033 * @s: SSL connection

2037 * is compatible with the client extensions.

2055 return 0;
2056}

2726/*
2727 * Return the number of security bits for the signature algorithm, or 0 on
2728 * error.
2729 */
2730static int sigalg_security_bits(SSL_CTX *ctx, const SIGALG_LOOKUP *lu)
2731{
2732 const EVP_MD *md = NULL((void*)0);
2733 int secbits = 0;
2734
2735 if (!tls1_lookup_md(ctx, lu, &md))
2736 return 0;
2737 if (md != NULL((void*)0)) {
2738 int md_type = EVP_MD_get_type(md);
2739
2740 /* Security bits: half digest bits */
2741 secbits = EVP_MD_get_size(md) * 4;
2742 if (secbits <= 0)
2743 return 0;
2744 /*
2745 * SHA1 and MD5 are known to be broken. Reduce security bits so that
2746 * they're no longer accepted at security level 1. The real values don't
2747 * really matter as long as they're lower than 80, which is our
2748 * security level 1.
2749 * https://eprint.iacr.org/2020/014 puts a chosen-prefix attack for
2750 * SHA1 at 2^63.4 and MD5+SHA1 at 2^67.2
2751 * https://documents.epfl.ch/users/l/le/lenstra/public/papers/lat.pdf
2752 * puts a chosen-prefix attack for MD5 at 2^39.
2753 */
2754 if (md_type == NID_sha164)
2755 secbits = 64;
2756 else if (md_type == NID_md5_sha1114)
2757 secbits = 67;
2758 else if (md_type == NID_md54)
2759 secbits = 39;
2760 } else {
2761 /* Values from https://tools.ietf.org/html/rfc8032#section-8.5 */
2762 if (lu->sigalg == TLSEXT_SIGALG_ed255190x0807)
2763 secbits = 128;
2764 else if (lu->sigalg == TLSEXT_SIGALG_ed4480x0808)
2765 secbits = 224;
2766 }
2767 /*
2768 * For provider-based sigalgs we have secbits information available
2769 * in the (provider-loaded) sigalg_list structure
2770 */
2771 if ((secbits == 0) && (lu->sig_idx >= SSL_PKEY_NUM9)
2772 && ((lu->sig_idx - SSL_PKEY_NUM9) < (int)ctx->sigalg_list_len)) {
2773 secbits = ctx->sigalg_list[lu->sig_idx - SSL_PKEY_NUM9].secbits;
2774 }
2775 return secbits;
2776}

4107 */
