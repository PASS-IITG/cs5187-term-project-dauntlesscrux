22{
23 uint64_t max, min;
24 int sign;

455 * Epoch 2 is always the fixed DTLS 1.3 handshake epoch: no compliant peer
456 * ever sends application data there (unlike epoch 1's early data). A record
457 * that only authenticates because of that epoch's retained keys must never
458 * be delivered as application data -- retained *application* epochs (3+,
459 * from KeyUpdate recovery) are unaffected, since they can legitimately
460 * carry reordered application traffic.
461 */
462int dtls_prev_epoch_allows_type(const OSSL_RECORD_LAYER *crypto_rl, int type)
463{
464 return !(crypto_rl->epoch == 2 && type == SSL3_RT_APPLICATION_DATA23);
465}

467/*-
468 * Call this to get a new input record.
469 * It will return <= 0 if more data is needed, normally due to an error
470 * or non-blocking IO.
471 * When it finishes, one packet has been decoded and can be found in
472 * ssl->s3.rrec.type - is the type of record
473 * ssl->s3.rrec.data - data
474 * ssl->s3.rrec.length - number of bytes
475 */
476int dtls_get_more_records(OSSL_RECORD_LAYER *rl)
477{
478 int rret;
479 size_t more, nread = 0;
480 TLS_RL_RECORD *rr;
481 DTLS_BITMAP *bitmap;
482 unsigned int is_next_epoch;
483 unsigned char recseqnum[6];
484 size_t recseqnumlen = 0;
485 size_t rechdrlen = 0;
486 size_t recseqnumoffs = 0;
487 int buffered_record = 0;
488 OSSL_RECORD_LAYER *crypto_rl;
489
490 rl->num_recs = 0;
491 rl->curr_rec = 0;
492 rl->num_released = 0;
493
494 rr = rl->rrec;
495
496 if (rl->rbuf.buf == NULL((void*)0)) {
497 if (!tls_setup_read_buffer(rl)) {
498 /* RLAYERfatal() already called */
499 return OSSL_RECORD_RETURN_FATAL-2;
500 }
501 }
502
503again:
504 crypto_rl = rl;
505 memset(recseqnum, 0, sizeof(recseqnum));
506
507 /* get something from the wire */
508
509 /* check if we have the header */
510 if (rl->rstate != SSL_ST_READ_BODY0xF1
511 || rl->packet_length < DTLS1_RT_HEADER_LENGTH13) {
512 PACKET dtlsrecord;
513 unsigned int record_type, record_version, epoch, length;
514 uint64_t epoch64;
515
516 /*
517 * If we have buffered records and the original BIO READ has all been processed
518 * let's leave and allow the Record Layer to update.
519 */
520 if (rl->version == DTLS1_3_VERSION0xFEFC
521 && buffered_record == 1 && rl->rbuf.left == 0) {
522 return OSSL_RECORD_RETURN_RETRY0;
523 }
524
525 rret = rl->funcs->read_n(rl, DTLS1_RT_HEADER_LENGTH13,
526 TLS_BUFFER_get_len(&rl->rbuf)((&rl->rbuf)->len), 0, 1, &nread);
527 /* read timeout is handled by dtls1_read_bytes */
528 if (rret < OSSL_RECORD_RETURN_SUCCESS1) {
529 /* RLAYERfatal() already called if appropriate */
530 return rret; /* error or non-blocking */
531 }
532
533 /* this packet contained a partial record, dump it */
534 if (rl->packet_length != DTLS1_RT_HEADER_LENGTH13) {
535 rl->packet_length = 0;
536 goto again;
537 }
538
539 rl->rstate = SSL_ST_READ_BODY0xF1;
540
541 if (!PACKET_buf_init(&dtlsrecord, rl->packet, rl->packet_length)
542 || !PACKET_get_1(&dtlsrecord, &record_type)) {
543 rl->packet_length = 0;
544 goto again;
545 }
546
547 /* Pull apart the header into the DTLS1_RECORD */
548 rr->type = (int)record_type;
549
550 /*-
551 * rfc9147:
552 * Implementations can demultiplex DTLS 1.3 records by examining the first
553 * byte as follows:
554 * * If the first byte is alert(21), handshake(22), or ack(proposed, 26),
555 * the record MUST be interpreted as a DTLSPlaintext record.
556 * * If the first byte is any other value, then receivers MUST check to
557 * see if the leading bits of the first byte are 001. If so, the implementation
558 * MUST process the record as DTLSCiphertext; the true content type
559 * will be inside the protected portion.
560 * * Otherwise, the record MUST be rejected as if it had failed deprotection,
561 * as described in Section 4.5.2.
562 */
563 if (rl->version == DTLS1_3_VERSION0xFEFC
564 && rr->type != SSL3_RT_ALERT21
565 && rr->type != SSL3_RT_HANDSHAKE22
566 && rr->type != SSL3_RT_ACK26
567 && !DTLS13_UNI_HDR_FIX_BITS_IS_SET(rr->type)(((rr->type) & 0xe0) == 0x20)) {
568 /* Silently discard */
569 rr->length = 0;
570 rl->packet_length = 0;
571 goto again;
572 }
573
574 if (DTLS13_UNI_HDR_FIX_BITS_IS_SET(rr->type)(((rr->type) & 0xe0) == 0x20)) {
575 /*
576 * rfc9147:
577 * receivers MUST check to if the leading bits of the first byte are 001.
578 * If so, the implementation MUST process the record as DTLSCiphertext;
579 */
580 int cbitisset = DTLS13_UNI_HDR_CID_BIT_IS_SET(rr->type)(((rr->type) & 0x10) == 0x10);
581 int sbitisset = DTLS13_UNI_HDR_SEQ_BIT_IS_SET(rr->type)(((rr->type) & 0x08) == 0x08);
582 int lbitisset = DTLS13_UNI_HDR_LEN_BIT_IS_SET(rr->type)(((rr->type) & 0x04) == 0x04);
583 uint16_t eebits = rr->type & DTLS13_UNI_HDR_EPOCH_BITS_MASK0x03;
584
585 record_version = DTLS1_2_VERSION0xFEFD;
586 epoch64 = rl->epoch;
587 recseqnumlen = sbitisset ? 2 : 1;
588 recseqnumoffs = sizeof(recseqnum) - recseqnumlen;
589
590 if (/* OpenSSL does not support connection IDs: silently discard */
591 cbitisset
592 /*
593 * Naive approach? We expect sequence number to be filled already
594 * and then override the last bytes of the sequence number.
595 */
596 || !PACKET_copy_bytes(&dtlsrecord, recseqnum + recseqnumoffs, recseqnumlen)
597 /*
598 * rfc9147:
599 * The length field MAY be omitted by clearing the L bit, which means
600 * that the record consumes the entire rest of the datagram in the
601 * lower level transport
602 */
603 || (lbitisset ? !PACKET_get_net_2(&dtlsrecord, &length)
604 : (length = (unsigned int)TLS_BUFFER_get_len(&rl->rbuf)((&rl->rbuf)->len)) > 0)) {
605 rr->length = 0;
606 rl->packet_length = 0;
607 goto again;
608 }
609
610 /*
611 * RFC 9147 Section 4.2.2 Says that after the handshake phase (epoch 3+)
612 * if the epoch bits do not match the current epoch we should use the
613 * last epoch value. That would result in dropping the packet.
614 */
615 if ((epoch64 & DTLS13_UNI_HDR_EPOCH_BITS_MASK0x03) != eebits) {
616 /*
617 * Since we do not transition out of epoch 0 or early data until after
618 * we receive the next record if we are in epoch 0 or Early Data (epoch 1)
619 * and get an epoch 2 record we must update the epoch
620 */
621 if (eebits == 2 && (epoch64 == 1 || epoch64 == 0)) {
622 epoch64 = 2;
623 } else if (rl->prev_epoch_rl != NULL((void*)0)
624 && (rl->prev_epoch_rl->epoch
625 & DTLS13_UNI_HDR_EPOCH_BITS_MASK0x03)
626 == eebits) {
627 /*
628 * This may be a retransmission at the epoch we have just
629 * moved on from, sent because the ACK we gave it was
630 * lost. Authenticate it with that epoch's own retained
631 * keys and replay window instead of dropping it.
632 */
633 epoch64 = rl->prev_epoch_rl->epoch;
634 crypto_rl = rl->prev_epoch_rl;
635 } else {
636 rr->length = 0;
637 rl->packet_length = 0;
638 goto again;
639 }
640 }
641 } else {
642 if (!PACKET_get_net_2(&dtlsrecord, &record_version)
643 || !PACKET_get_net_2(&dtlsrecord, &epoch)
644 || !PACKET_copy_bytes(&dtlsrecord, recseqnum, 6)
645 || !PACKET_get_net_2(&dtlsrecord, &length)) {
646 rr->length = 0;
647 rl->packet_length = 0;
648 goto again;
649 }
650 epoch64 = epoch;
651 recseqnumoffs = 0;
652 recseqnumlen = 6;
653 }
654
655 rechdrlen = PACKET_data(&dtlsrecord) - rl->packet;
656 rr->rec_version = (int)record_version;
657 rr->epoch = epoch64;
658 rr->length = length;
659
660 if (rl->msg_callback != NULL((void*)0))
661 rl->msg_callback(0, rr->rec_version, SSL3_RT_HEADER0x100, rl->packet,
662 rechdrlen, rl->cbarg);
663
664 /*
665 * Lets check the version. We tolerate alerts that don't have the exact
666 * version number (e.g. because of protocol version errors)
667 */
668 if (!rl->is_first_record && rr->type != SSL3_RT_ALERT21
669 /* DTLSv1.3 records sets the legacy version field to DTLSv1.2 */
670 && !(rr->rec_version == DTLS1_2_VERSION0xFEFD
671 && rl->version == DTLS1_3_VERSION0xFEFC)) {
672 if (rr->rec_version != rl->version) {
673 /* unexpected version, silently discard */
674 rr->length = 0;
675 rl->packet_length = 0;
676 goto again;
677 }
678 }
679
680 if (rr->rec_version >> 8 != (rl->version == DTLS_ANY_VERSION0x1FFFF ? DTLS1_VERSION_MAJOR0xFE : rl->version >> 8)) {
681 /* wrong version, silently discard record */
682 rr->length = 0;
683 rl->packet_length = 0;
684 goto again;
685 }
686
687 if (rr->length > SSL3_RT_MAX_ENCRYPTED_LENGTH((256 + 64) + (16384 + 1024))) {
688 /* record too long, silently discard it */
689 rr->length = 0;
690 rl->packet_length = 0;
691 goto again;
692 }
693
694 /*
695 * If received packet overflows maximum possible fragment length then
696 * silently discard it
697 */
698 if (rr->length > rl->max_frag_len + SSL3_RT_MAX_ENCRYPTED_OVERHEAD(256 + 64)) {
699 /* record too long, silently discard it */
700 rr->length = 0;
701 rl->packet_length = 0;
702 goto again;
703 }
704
705 /* now rl->rstate == SSL_ST_READ_BODY */
706 }
707
708 /* rl->rstate == SSL_ST_READ_BODY, get and decode the data */
709
710 if (rr->length > rl->packet_length - DTLS1_RT_HEADER_LENGTH13) {
711 /* now rl->packet_length == DTLS1_RT_HEADER_LENGTH */
712 more = rr->length - (nread - rechdrlen);
713 rret = rl->funcs->read_n(rl, more, more, 1, 1, &nread);
714 /* this packet contained a partial record, dump it */
715 if (rret < OSSL_RECORD_RETURN_SUCCESS1 || nread != more) {
716 if (rl->alert != SSL_AD_NO_ALERT-1) {
717 /* read_n() called RLAYERfatal() */
718 return OSSL_RECORD_RETURN_FATAL-2;
719 }
720 rr->length = 0;
721 rl->packet_length = 0;
722 goto again;
723 }
724
725 /*
726 * now n == rr->length,
727 * and rl->packet_length == DTLS1_RT_HEADER_LENGTH + rr->length
728 */
729 }
730 /* set state for later operations */
731 rl->rstate = SSL_ST_READ_HEADER0xF0;
732
733 /*
734 * RFC 9147 permits DTLSPlaintext only in epoch 0. Silently discard it
735 * after reading the fragment so later records remain framed.
736 */
737 if (rl->version == DTLS1_3_VERSION0xFEFC
738 && rl->epoch != 0
739 && !DTLS13_UNI_HDR_FIX_BITS_IS_SET(rr->type)(((rr->type) & 0xe0) == 0x20)) {
740 rr->length = 0;
741 rl->packet_length = 0;
742 goto again;
743 }
744
745 /*
746 * rfc9147:
747 * This procedure requires the ciphertext length to be at least 16 bytes.
748 * Receivers MUST reject shorter records as if they had failed deprotection
749 */
750 if (DTLS13_UNI_HDR_FIX_BITS_IS_SET(rr->type)(((rr->type) & 0xe0) == 0x20)
751 && rl->version == DTLS1_3_VERSION0xFEFC
752 && !(rl->in_init && rl->epoch == 0)
753 && ((rl->packet_length < rechdrlen + DTLS13_CIPHERTEXT_MINSIZE16)
754 || (crypto_rl->sn_enc_ctx == NULL((void*)0) && crypto_rl->mac_ctx == NULL((void*)0))
755 || (crypto_rl->sn_enc_ctx != NULL((void*)0)
756 && !dtls_crypt_sequence_number(crypto_rl->sn_enc_ctx,
757 recseqnum + recseqnumoffs,
758 recseqnumlen,
759 rl->packet + rechdrlen)))) {
760 /* sequence number encryption failed dump record */
761 rr->length = 0;
762 rl->packet_length = 0;
763 goto again;
764 }
765
766 if (rl->version == DTLS1_3_VERSION0xFEFC && rr->epoch == crypto_rl->epoch
767 && DTLS13_UNI_HDR_FIX_BITS_IS_SET(rr->type)(((rr->type) & 0xe0) == 0x20)) {
768 /* Reconstruct current-epoch unified records using its replay window. */
769 uint64_t truncated = 0;
770 size_t i;
771
772 /* A unified header carries 8 or 16 bits of the sequence number */
773 if (!ossl_assert(recseqnumlen == 1 || recseqnumlen == 2)__builtin_expect(!!((recseqnumlen == 1 || recseqnumlen == 2) !=
0), 1)) {
774 RLAYERfatal(rl, SSL_AD_INTERNAL_ERROR, ERR_R_INTERNAL_ERROR)(ERR_new(), ERR_set_debug("ssl/record/methods/dtls_meth.c", 774
, __func__), ossl_rlayer_fatal)((rl), (80), ((259 | ((0x1 <<
18L) | (0x2 << 18L)))), ((void*)0));
775 return OSSL_RECORD_RETURN_FATAL-2;
776 }
777
778 for (i = 0; i < recseqnumlen; i++)
779 truncated = (truncated << 8) | recseqnum[recseqnumoffs + i];
780
781 crypto_rl->sequence = dtls13_reconstruct_seq_num(crypto_rl->bitmap.max_seq_num,
782 truncated, recseqnumlen);
783 } else {
784 /*
785 * DTLSPlaintext carries 48 bits. A buffered next-epoch unified record
786 * is re-parsed after that epoch's record layer is installed, so this
787 * provisional value is unused.
788 */
789 rl->sequence = ((uint64_t)recseqnum[0]) << 40;
790 rl->sequence |= ((uint64_t)recseqnum[1]) << 32;
791 rl->sequence |= ((uint64_t)recseqnum[2]) << 24;
792 rl->sequence |= ((uint64_t)recseqnum[3]) << 16;
793 rl->sequence |= ((uint64_t)recseqnum[4]) << 8;
794 rl->sequence |= ((uint64_t)recseqnum[5]) << 0;
795 }
796
797 /* match epochs. NULL means the packet is dropped on the floor */
798 bitmap = dtls_get_bitmap(crypto_rl, rr, &is_next_epoch);
799 if (bitmap == NULL((void*)0) && !is_next_epoch) {
800 rr->length = 0;
801 rl->packet_length = 0; /* dump this record */
802 goto again; /* get another record */
803 }
804
805 /*
806 * If this record is from the next epoch (either HM or ALERT), and a
807 * handshake is currently in progress, buffer it since it cannot be
808 * processed at this time.
809 */
810 if (is_next_epoch) {
811 /*
812 * DTLS 1.3 has two scenarios when to buffer the record
813 * from the next epoch
814 * 1) during handshake when rl->in_init is set and
815 * we got an epoch 0 record and the record layer
816 * is at epoch 2
817 * 2) during early data when rl-epoch is 1 and we
818 * got an epoch 2 record
819 */
820 if ((rl->in_init && rl->version != DTLS1_3_VERSION0xFEFC)
821 || (rl->in_init && rl->epoch == 0 && rr->epoch == 2)
822 || (rl->version == DTLS1_3_VERSION0xFEFC
823 && ((rl->in_init && rl->epoch == 0 && rr->epoch == 2)
824 || (rl->epoch == 1 && rr->epoch == 2)))) {
825
826 uint64_t unprocessed_record_priority = rr->seq_num;
827
828 /*
829 * DTLS1.3 uses encrypted sequence numbers so we want to set the
830 * priority base off of epoch.
831 */
832 if (rl->version == DTLS1_3_VERSION0xFEFC || (rr->epoch == 2 && rl->epoch == 0)) {
833 if (rr->epoch == 1)
834 unprocessed_record_priority = rl->dtls13_epoch_1_seq++;
835 else if (rr->epoch == 2)
836 unprocessed_record_priority = rl->dtls13_epoch_2_seq++;
837 }
838 if (dtls_rlayer_buffer_record(rl, &rl->unprocessed_rcds,
839 unprocessed_record_priority)
840 < 0) {
841 /* RLAYERfatal() already called */
842 return OSSL_RECORD_RETURN_FATAL-2;
843 }
844 buffered_record = 1;
845 }
846 rr->length = 0;
847 rl->packet_length = 0;
848 goto again;
849 }
850
851 /*
852 * dtls_record_replay_check() and dtls_process_record() read and write
853 * whichever OSSL_RECORD_LAYER they are passed, not rr. When crypto_rl is
854 * the retained previous epoch, seed its record slot from rr so they
855 * have this record's data to work with; a no-op when crypto_rl == rl.
856 */
857 if (crypto_rl != rl) {
858 crypto_rl->packet = rl->packet;
859 crypto_rl->packet_length = rl->packet_length;
860 crypto_rl->rrec[0] = *rr;
861 }
862
863#ifndef OPENSSL_NO_SCTP
864 /* Only do replay check if no SCTP bio (also check for NULL bio) */
865 if (rl->bio == NULL((void*)0) || !BIO_dgram_is_sctp(rl->bio)) {
866#endif
867 /* Check whether this is a repeat, or aged record. */
868 if (!dtls_record_replay_check(crypto_rl, bitmap)) {
869 rr->length = 0;
870 rl->packet_length = 0; /* dump this record */
871 goto again; /* get another record */
872 }
873#ifndef OPENSSL_NO_SCTP
874 }
875#endif
876
877 /* just read a 0 length packet */
878 if (rr->length == 0)
879 goto again;
880
881 if (!dtls_process_record(crypto_rl, bitmap)) {
882 if (crypto_rl->alert != SSL_AD_NO_ALERT-1) {
883 /* dtls_process_record() called RLAYERfatal */
884 rl->alert = crypto_rl->alert;
885 return OSSL_RECORD_RETURN_FATAL-2;
886 }
887 rr->length = 0;
888 rl->packet_length = 0; /* dump this record */
889 goto again; /* get another record */
890 }
891
892 if (crypto_rl != rl) {
893 *rr = crypto_rl->rrec[0];
894 rl->packet_length = 0;
895 }
896
897 if (rl->funcs->post_process_record && !rl->funcs->post_process_record(rl, rr)) {
898 /* RLAYERfatal already called */
899 return OSSL_RECORD_RETURN_FATAL-2;
900 }
901
902 /* rr->type is only decoded after post_process_record() above. */
903 if (crypto_rl != rl && !dtls_prev_epoch_allows_type(crypto_rl, rr->type)) {
904 rr->length = 0;
905 goto again;
906 }
907
908 if (rr->length == 0) {
909 /* No payload data in this record. Dump it */
910 rl->packet_length = 0;
911 goto again;
912 }
913
914 rl->num_recs = 1;
915 return OSSL_RECORD_RETURN_SUCCESS1;
916}

918/*
919 * Push any unread buffered bytes and any records already buffered in
920 * unprocessed_rcds (see is_next_epoch in dtls_get_more_records()) forward
921 * into rl->next, so a pending handshake epoch transition still completes.
922 * Must run when rl stops being the active read layer, not deferred until
923 * it is eventually freed -- see dtls_free() and dtls_set_prev_epoch_rl().
924 */
925static int dtls_forward_pending_records(OSSL_RECORD_LAYER *rl)
926{
927 TLS_BUFFER *rbuf = &rl->rbuf;
928 size_t left, written;
929 pitem *item;
930 DTLS_RLAYER_RECORD_DATA *rdata;
931 int ret = 1;
932
933 left = rbuf->left;
934 if (left > 0) {
935 ret = BIO_write_ex(rl->next, rbuf->buf + rbuf->offset, left, &written);
936 rbuf->left = 0;
937 }
938
939 while ((item = pqueue_pop(&rl->unprocessed_rcds)) != NULL((void*)0)) {
940 rdata = (DTLS_RLAYER_RECORD_DATA *)item->data;
941
942 ret &= BIO_write_ex(rl->next, rdata->packet, rdata->packet_length,
943 &written);
944 OPENSSL_free(rdata->packet)CRYPTO_free(rdata->packet, "ssl/record/methods/dtls_meth.c"
, 944);
945 OPENSSL_free(item->data)CRYPTO_free(item->data, "ssl/record/methods/dtls_meth.c", 945
);
946 pitem_free(item);
947 }
948
949 return ret;
950}

952static int dtls_free(OSSL_RECORD_LAYER *rl)
953{
954 int ret = dtls_forward_pending_records(rl);
955
956 if (rl->prev_epoch_rl != NULL((void*)0)) {
957 ret &= dtls_free(rl->prev_epoch_rl);
958 rl->prev_epoch_rl = NULL((void*)0);
959 }
960
961 return tls_free(rl) && ret;
962}

964/*
965 * Take ownership of the previous record layer, just superseded by the
966 * record layer, instead of the caller freeing it.
967 */
968static int dtls_set_prev_epoch_rl(OSSL_RECORD_LAYER *rl, OSSL_RECORD_LAYER *prev)
969{
970 int ret = dtls_forward_pending_records(prev);
971
972 if (prev->prev_epoch_rl != NULL((void*)0)) {
973 ret &= dtls_free(prev->prev_epoch_rl);
974 prev->prev_epoch_rl = NULL((void*)0);
975 }
976
977 /*
978 * prev's own read buffer is never used again: authenticating a
979 * retransmission at this epoch reuses the active layer's packet buffer
980 * (see the crypto_rl handling in dtls_get_more_records()). Release it
981 * now instead of leaving it allocated until the whole layer chain is
982 * torn down or the caller happens to call SSL_free_buffers().
983 */
984 ret &= tls_release_read_buffer(prev);
985
986 rl->prev_epoch_rl = prev;
987 return ret;
988}

1144static size_t dtls_get_max_record_overhead(OSSL_RECORD_LAYER *rl)
1145{
1146 size_t blocksize = 0, contenttypelen = 0;
1147 size_t rchdrlen = tls_get_record_header_len(rl);
1148
1149 if (rl->enc_ctx != NULL((void*)0) && (EVP_CIPHER_CTX_get_mode(rl->enc_ctx)EVP_CIPHER_get_mode(EVP_CIPHER_CTX_get0_cipher(rl->enc_ctx
)) == EVP_CIPH_CBC_MODE0x2))
1150 blocksize = EVP_CIPHER_CTX_get_block_size(rl->enc_ctx);
1151
1152 /* DTLSv1.3 adds an extra content type byte after payload data */
1153 if (rl->version == DTLS1_3_VERSION0xFEFC)
1154 contenttypelen = 1;
1155
1156 /*
1157 * If we have a cipher in place then the tag is mandatory. If the cipher is
1158 * CBC mode then an explicit IV is also mandatory. If we know the digest,
1159 * then we check it is consistent with the taglen. In the case of stitched
1160 * ciphers or AEAD ciphers we don't now the digest (or there isn't one) so
1161 * we just trust that the taglen is correct.
1162 */
1163 assert(rl->enc_ctx == NULL || ((blocksize == 0 || rl->eivlen > 0) && rl->taglen > 0))((void) (0));
1164 assert(rl->md == NULL || (int)rl->taglen == EVP_MD_size(rl->md))((void) (0));
1165
1166 /*
1167 * Record overhead consists of the record header, the explicit IV, any
1168 * expansion due to cbc padding, and the mac/tag len. There could be
1169 * further expansion due to compression - but we don't know what this will
1170 * be without knowing the length of the data. However when this function is
1171 * called we don't know what the length will be yet - so this is a catch-22.
1172 * We *could* use SSL_3_RT_MAX_COMPRESSED_OVERHEAD which is an upper limit
1173 * for the maximum record size. But this value is larger than our fallback
1174 * MTU size - so isn't very helpful. We just ignore potential expansion
1175 * due to compression.
1176 */
1177 return rchdrlen + rl->eivlen + blocksize + rl->taglen + contenttypelen;
1178}
