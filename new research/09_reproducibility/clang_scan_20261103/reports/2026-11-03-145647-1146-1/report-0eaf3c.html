1051 * Very basic selection callback: just use any certificate chain reported as
1052 * valid. More sophisticated could prioritise according to local policy.
1053 */
1054static int set_cert_cb(SSL *ssl, void *arg)
1055{
1056 int i, rv;
1057 SSL_EXCERT *exc = arg;
1058#ifdef CERT_CB_TEST_RETRY
1059 static int retry_cnt;
1060
1061 if (retry_cnt < 5) {
1062 retry_cnt++;
1063 BIO_printf(bio_err,
1064 "Certificate callback retry test: count %d\n",
1065 retry_cnt);

1067 }
1068#endif
1069 SSL_certs_clear(ssl);
1070
1071 if (exc == NULL((void*)0))
1072 return 1;

1074 /*
1075 * Go to end of list and traverse backwards since we prepend newer
1076 * entries this retains the original order.
1077 */
1078 while (exc->next != NULL((void*)0))
1079 exc = exc->next;
1080
1081 i = 0;
1082
1083 while (exc != NULL((void*)0)) {
1084 i++;
1085 rv = SSL_check_chain(ssl, exc->cert, exc->key, exc->chain);
1086 BIO_printf(bio_err, "Checking cert chain %d:\nSubject: ", i);
1087 X509_NAME_print_ex(bio_err, X509_get_subject_name(exc->cert), 0,
1088 get_nameopt());

1090 print_chain_flags(ssl, rv);
1091 if (rv & CERT_PKEY_VALID0x1) {
1092 if (!SSL_use_certificate(ssl, exc->cert)
1093 || !SSL_use_PrivateKey(ssl, exc->key)) {
1094 return 0;
1095 }
1096 /*
