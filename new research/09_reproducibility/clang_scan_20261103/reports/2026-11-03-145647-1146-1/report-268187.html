4114static int ldap_ExtendedResponse_parse(const char *buf, long rem)
4115{
4116 const unsigned char *cur, *end;
4117 long len;
4118 int tag, xclass, inf, ret = -1;
4119
4120 cur = (const unsigned char *)buf;
4121 end = cur + rem;
4122
4123 /*
4124 * From RFC 4511:
4125 *
4126 * LDAPMessage ::= SEQUENCE {
4127 * messageID MessageID,
4128 * protocolOp CHOICE {
4129 * ...
4130 * extendedResp ExtendedResponse,
4131 * ... },
4132 * controls [0] Controls OPTIONAL }
4133 *
4134 * ExtendedResponse ::= [APPLICATION 24] SEQUENCE {
4135 * COMPONENTS OF LDAPResult,
4136 * responseName [10] LDAPOID OPTIONAL,
4137 * responseValue [11] OCTET STRING OPTIONAL }
4138 *
4139 * LDAPResult ::= SEQUENCE {
4140 * resultCode ENUMERATED {
4141 * success (0),
4142 * ...
4143 * other (80),
4144 * ... },
4145 * matchedDN LDAPDN,
4146 * diagnosticMessage LDAPString,
4147 * referral [3] Referral OPTIONAL }
4148 */
4149
4150 /* pull SEQUENCE */
4151 inf = ASN1_get_object(&cur, &len, &tag, &xclass, rem);
4152 if (inf != V_ASN1_CONSTRUCTED0x20 || tag != V_ASN1_SEQUENCE16 || (rem = (long)(end - cur), len > rem)) {
4153 BIO_puts(bio_err, "Unexpected LDAP response\n");
4154 goto end;
4155 }
4156
4157 rem = len; /* ensure that we don't overstep the SEQUENCE */
4158
4159 /* pull MessageID */
4160 inf = ASN1_get_object(&cur, &len, &tag, &xclass, rem);
4161 if (inf != V_ASN1_UNIVERSAL0x00 || tag != V_ASN1_INTEGER2 || (rem = (long)(end - cur), len > rem)) {
4162 BIO_puts(bio_err, "No MessageID\n");
4163 goto end;
4164 }
4165
4166 cur += len; /* shall we check for MessageId match or just skip? */
4167
4168 /* pull [APPLICATION 24] */
4169 rem = (long)(end - cur);
4170 inf = ASN1_get_object(&cur, &len, &tag, &xclass, rem);
4171 if (inf != V_ASN1_CONSTRUCTED0x20 || xclass != V_ASN1_APPLICATION0x40 || tag != 24) {
4172 BIO_puts(bio_err, "Not ExtendedResponse\n");
4173 goto end;
4174 }
4175
4176 /* pull resultCode */
4177 rem = (long)(end - cur);
4178 inf = ASN1_get_object(&cur, &len, &tag, &xclass, rem);
4179 if (inf != V_ASN1_UNIVERSAL0x00 || tag != V_ASN1_ENUMERATED10 || len == 0 || (rem = (long)(end - cur), len > rem)) {
4180 BIO_puts(bio_err, "Not LDAPResult\n");
4181 goto end;
4182 }
4183
4184 /* len should always be one, but just in case... */
4185 for (ret = 0, inf = 0; inf < len; inf++) {
4186 ret <<= 8;
4187 ret |= cur[inf];
4188 }
4189 /* There is more data, but we don't care... */
4190end:
4191 return ret;
4192}

4194/*
4195 * Host dNS Name verifier: used for checking that the hostname is in dNS format
4196 * before setting it as SNI
4197 */
4198static int is_dNS_name(const char *host)
4199{
4200 const size_t MAX_LABEL_LENGTH = 63;
4201 size_t i;
4202 int isdnsname = 0;
4203 size_t length = strlen(host);
4204 size_t label_length = 0;
4205 int all_numeric = 1;
4206
4207 /*
4208 * Deviation from strict DNS name syntax, also check names with '_'
4209 * Check DNS name syntax, any '-' or '.' must be internal,
4210 * and on either side of each '.' we can't have a '-' or '.'.
4211 *
4212 * If the name has just one label, we don't consider it a DNS name.
4213 */
4214 for (i = 0; i < length && label_length < MAX_LABEL_LENGTH; ++i) {
4215 char c = host[i];
4216
4217 if ((c >= 'a' && c <= 'z')
4218 || (c >= 'A' && c <= 'Z')
4219 || c == '_') {
4220 label_length += 1;
4221 all_numeric = 0;
4222 continue;
4223 }
4224
4225 if (c >= '0' && c <= '9') {
4226 label_length += 1;
4227 continue;
4228 }
4229
4230 /* Dot and hyphen cannot be first or last. */
4231 if (i > 0 && i < length - 1) {
4232 if (c == '-') {
4233 label_length += 1;
4234 continue;
4235 }
4236 /*
4237 * Next to a dot the preceding and following characters must not be
4238 * another dot or a hyphen. Otherwise, record that the name is
4239 * plausible, since it has two or more labels.
4240 */
4241 if (c == '.'
4242 && host[i + 1] != '.'
4243 && host[i - 1] != '-'
4244 && host[i + 1] != '-') {
4245 label_length = 0;
4246 isdnsname = 1;
4247 continue;
4248 }
4249 }
4250 isdnsname = 0;
4251 break;
4252 }
4253
4254 /* dNS name must not be all numeric and labels must be shorter than 64 characters. */
4255 isdnsname &= !all_numeric && !(label_length == MAX_LABEL_LENGTH);
4256
4257 return isdnsname;
4258}

4367static int user_data_process(struct user_data_st *user_data, size_t *len,
4368 size_t *off)
4369{
4370 char *buf_start = user_data->buf + user_data->bufoff;
4371 size_t outlen = user_data->buflen;
4372 const int version1_3 = SSL_is_dtls(user_data->con) ? DTLS1_3_VERSION0xFEFC : TLS1_3_VERSION0x0304;
4373
4374 if (user_data->buflen == 0) {
4375 *len = 0;
4376 *off = 0;
4377 return USER_DATA_PROCESS_NO_DATA3;
4378 }
4379
4380 if (user_data->mode == USER_DATA_MODE_BASIC1) {
4381 switch (buf_start[0]) {
4382 case 'Q':
4383 user_data->buflen = user_data->bufoff = *len = *off = 0;
4384 return user_data_execute(user_data, USER_COMMAND_QUIT1, NULL((void*)0));
4385
4386 case 'C':
4387 user_data->buflen = user_data->bufoff = *len = *off = 0;
4388 return user_data_execute(user_data, USER_COMMAND_RECONNECT2, NULL((void*)0));
4389
4390 case 'R':
4391 user_data->buflen = user_data->bufoff = *len = *off = 0;
4392 return user_data_execute(user_data, USER_COMMAND_RENEGOTIATE3, NULL((void*)0));
4393
4394 case 'K':
4395 case 'k':
4396 user_data->buflen = user_data->bufoff = *len = *off = 0;
4397 return user_data_execute(user_data, USER_COMMAND_KEY_UPDATE4,
4398 buf_start[0] == 'K' ? "req" : "noreq");
4399 default:
4400 break;
4401 }
4402 } else if (user_data->mode == USER_DATA_MODE_ADVANCED2) {
4403 char *cmd_start = buf_start;
4404
4405 cmd_start[outlen] = '\0';
4406 for (;;) {
4407 cmd_start = strchr(cmd_start, '{');
4408 if (cmd_start == buf_start && *(cmd_start + 1) == '{') {
4409 /* The "{" is escaped, so skip it */
4410 cmd_start += 2;
4411 buf_start++;
4412 user_data->bufoff++;
4413 user_data->buflen--;
4414 outlen--;
4415 continue;
4416 }
4417 break;
4418 }
4419
4420 if (cmd_start == buf_start) {
4421 /* Command detected */
4422 char *cmd_end = strchr(cmd_start, '}');
4423 char *arg_start;
4424 int cmd = -1, ret = USER_DATA_PROCESS_NO_DATA3;
4425 size_t oldoff;
4426
4427 if (cmd_end == NULL((void*)0)) {
4428 /* Malformed command */
4429 cmd_start[outlen - 1] = '\0';
4430 BIO_printf(bio_err,
4431 "ERROR PROCESSING COMMAND. REST OF LINE IGNORED: %s\n",
4432 cmd_start);
4433 user_data->buflen = user_data->bufoff = *len = *off = 0;
4434 return USER_DATA_PROCESS_NO_DATA3;
4435 }
4436 *cmd_end = '\0';
4437 arg_start = strchr(cmd_start, ':');
4438 if (arg_start != NULL((void*)0)) {
4439 *arg_start = '\0';
4440 arg_start++;
4441 }
4442 /* Skip over the { */
4443 cmd_start++;
4444 /*
4445 * Now we have cmd_start pointing to a NUL terminated string for
4446 * the command, and arg_start either being NULL or pointing to a
4447 * NUL terminated string for the argument.
4448 */
4449 if (OPENSSL_strcasecmp(cmd_start, "help") == 0) {
4450 cmd = USER_COMMAND_HELP0;
4451 } else if (OPENSSL_strcasecmp(cmd_start, "quit") == 0) {
4452 cmd = USER_COMMAND_QUIT1;
4453 } else if (OPENSSL_strcasecmp(cmd_start, "reconnect") == 0) {
4454 cmd = USER_COMMAND_RECONNECT2;
4455 } else if (SSL_is_quic(user_data->con)) {
4456 if (OPENSSL_strcasecmp(cmd_start, "fin") == 0)
4457 cmd = USER_COMMAND_FIN5;
4458 }
4459 if (SSL_version(user_data->con) == version1_3) {
4460 if (OPENSSL_strcasecmp(cmd_start, "keyup") == 0) {
4461 cmd = USER_COMMAND_KEY_UPDATE4;
4462 if (arg_start == NULL((void*)0))
4463 arg_start = "req";
4464 }
4465 } else {
4466 /* (D)TLSv1.2 or below */
4467 if (OPENSSL_strcasecmp(cmd_start, "reneg") == 0)
4468 cmd = USER_COMMAND_RENEGOTIATE3;
4469 }
4470 if (cmd == -1) {
4471 BIO_printf(bio_err, "UNRECOGNISED COMMAND (IGNORED): %s\n",
4472 cmd_start);
4473 } else {
4474 ret = user_data_execute(user_data, cmd, arg_start);
4475 if (ret == USER_DATA_PROCESS_BAD_ARGUMENT0) {
4476 BIO_printf(bio_err, "BAD ARGUMENT (COMMAND IGNORED): %s\n",
4477 arg_start);
4478 ret = USER_DATA_PROCESS_NO_DATA3;
4479 }
4480 }
4481 oldoff = user_data->bufoff;
4482 user_data->bufoff = (cmd_end - user_data->buf) + 1;
4483 user_data->buflen -= user_data->bufoff - oldoff;
4484 if (user_data->buf + 1 == cmd_start
4485 && user_data->buflen == 1
4486 && user_data->buf[user_data->bufoff] == '\n') {
4487 /*
4488 * This command was the only thing on the whole line. We
4489 * suppress the final `\n`
4490 */
4491 user_data->bufoff = 0;
4492 user_data->buflen = 0;
4493 }
4494 *len = *off = 0;
4495 return ret;
4496 } else if (cmd_start != NULL((void*)0)) {
4497 /*
4498 * There is a command on this line, but its not at the start. Output
4499 * the start of the line, and we'll process the command next time
4500 * we call this function
4501 */
4502 outlen = cmd_start - buf_start;
4503 }
4504 }
4505
4506 if (user_data->isfin) {
4507 user_data->buflen = user_data->bufoff = *len = *off = 0;
4508 return USER_DATA_PROCESS_NO_DATA3;
4509 }
4510
4511#ifdef CHARSET_EBCDIC
4512 ebcdic2ascii(buf_start, buf_start, outlen);
4513#endif
4514 *len = outlen;
4515 *off = user_data->bufoff;
4516 user_data->buflen -= outlen;
4517 if (user_data->buflen == 0)
4518 user_data->bufoff = 0;
4519 else
4520 user_data->bufoff += outlen;
4521 return USER_DATA_PROCESS_CONTINUE4;
4522}
