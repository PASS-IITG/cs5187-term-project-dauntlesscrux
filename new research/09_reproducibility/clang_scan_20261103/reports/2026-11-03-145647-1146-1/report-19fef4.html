391/*
392 * We need to determine our addressing mode. There are basically two ways we can
393 * use L4 addresses:
394 *
395 * - Addressed mode, in which our BIO_sendmmsg calls have destination
396 * addresses attached to them which we expect the underlying network BIO to
397 * handle;
398 *
399 * - Unaddressed mode, in which the BIO provided to us on the network side
400 * neither provides us with L4 addresses nor is capable of honouring ones we
401 * provide. We don't know where the QUIC traffic we send ends up exactly and
402 * trust the application to know what it is doing.
403 *
404 * Addressed mode is preferred because it enables support for connection
405 * migration, multipath, etc. in the future. Addressed mode is automatically
406 * enabled if we are using e.g. BIO_s_datagram, with or without BIO_s_connect.
407 *
408 * If we are passed a BIO_s_dgram_pair (or some custom BIO) we may have to use

410 * provide and honour L4 addresses.

414 * need, we use addressed mode. Otherwise, we use unaddressed mode.

417 * this is not set, we fail. If unaddressed mode is used, we do not require

833/* Process incoming datagrams, if any. */
834static void port_rx_pre(QUIC_PORT *port)
835{
836 int ret;
837
838 /*
839 * Originally, this check (don't RX before we have sent anything if we are
840 * not a server, because there can't be anything) was just intended as a
841 * minor optimisation. However, it is actually required on Windows, and
842 * removing this check will cause Windows to break.
843 *
844 * The reason is that under Win32, recvfrom() does not work on a UDP socket
845 * which has not had bind() called (???). However, calling sendto() will
846 * automatically bind an unbound UDP socket. Therefore, if we call a Winsock
847 * recv-type function before calling a Winsock send-type function, that call
848 * will fail with WSAEINVAL, which we will regard as a permanent network
849 * error.
850 *
851 * Therefore, this check is essential as we do not require our API users to
852 * bind a socket first when using the API in client mode.
853 */
854 if (!port->allow_incoming && !port->have_sent_any_pkt)
855 return;
856
857 /*
858 * Get DEMUX to BIO_recvmmsg from the network and queue incoming datagrams
859 * to the appropriate QRX instances.
860 */
861 ret = ossl_quic_demux_pump(port->demux);
862 if (ret == QUIC_DEMUX_PUMP_RES_PERMANENT_FAIL(-2))
863 /*
864 * We don't care about transient failure, but permanent failure means we
865 * should tear down the port. All connections skip straight to the
866 * Terminated state as there is no point trying to send CONNECTION_CLOSE
867 * frames if the network BIO is not operating correctly.
868 */
869 ossl_quic_port_raise_net_error(port, NULL((void*)0));
870}

887 */

890 port->tserver_ch = NULL((void*)0);

949static int port_try_handle_stateless_reset(QUIC_PORT *port, const QUIC_URXE *e)
950{
951 size_t i;
952 const unsigned char *data = ossl_quic_urxe_data(e);
953 void *opaque = NULL((void*)0);
954
955 /*
956 * Perform some fast and cheap checks for a packet not being a stateless
957 * reset token. RFC 9000 s. 10.3 specifies this layout for stateless
958 * reset packets:
959 *
960 * Stateless Reset {
961 * Fixed Bits (2) = 1,
962 * Unpredictable Bits (38..),
963 * Stateless Reset Token (128),
964 * }
965 *
966 * It also specifies:
967 * However, endpoints MUST treat any packet ending in a valid
968 * stateless reset token as a Stateless Reset, as other QUIC
969 * versions might allow the use of a long header.
970 *
971 * We can rapidly check for the minimum length and that the first pair
972 * of bits in the first byte are 01 or 11.
973 *
974 * The function returns 1 if it is a stateless reset packet, 0 if it isn't
975 * and -1 if an error was encountered.
976 */
977 if (e->data_len < QUIC_STATELESS_RESET_TOKEN_LEN16 + 5
978 || (0100 & *data) != 0100)
979 return 0;
980
981 for (i = 0;; ++i) {
982 if (!ossl_quic_srtm_lookup(port->srtm,
983 (const QUIC_STATELESS_RESET_TOKEN *)(data + e->data_len
984 - sizeof(QUIC_STATELESS_RESET_TOKEN)),
985 i, &opaque, NULL((void*)0)))
986 break;
987
988 assert(opaque != NULL)((void) (0));
989 ossl_quic_channel_on_stateless_reset((QUIC_CHANNEL *)opaque);
990 }
991
992 return i > 0;
993}

1890 /*
1891 * Time to reinject packets from qrx to channel before
1892 * qrx will be destroyed here.
1893 */
1894 ossl_quic_tx_packetiser_add_unvalidated_credit(new_ch->txp, e->data_len);
1895 while (ossl_qrx_read_pkt(qrx_src, &qrx_pkt) == 1)
1896 ossl_quic_channel_inject_pkt(new_ch, qrx_pkt);
1897 ossl_qrx_update_pn_space(qrx_src, new_ch->qrx);
1898 /*
1899 * transfer ownership back to qrx;
1900 */
1901 qrx = qrx_src;
1902 qrx_src = NULL((void*)0);
1903 }
1904
1905 /*
1906 * If function reaches this place, then packet got validated in
1907 * ossl_qrx_validate_initial_packet(). Keep in mind the function
1908 * ossl_qrx_validate_initial_packet() decrypts the packet to validate it.
1909 * If packet validation was successful (and it was because we are here),
1910 * then the function puts the packet to qrx->rx_pending. We must not call
1911 * ossl_qrx_inject_urxe() here now, because we don't want to insert
1912 * the packet to qrx->urx_pending which keeps packet waiting for decryption.
1913 *
1914 * We are going to call ossl_quic_demux_release_urxe() to dispose buffer
1915 * which still holds encrypted data.
1916 */
1917
1918undesirable:
1919 ossl_qrx_free(qrx); /* releases reference */
1920 ossl_qrx_free(qrx_src);
1921 ossl_quic_demux_release_urxe(port->demux, e);
1922}
