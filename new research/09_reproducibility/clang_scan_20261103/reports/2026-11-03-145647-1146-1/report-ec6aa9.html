39/*
40 * Barrett reduction approximates (x mod q) without a division instruction.
41 * It pre-computes m = floor(2^BARRETT_SHIFT / q) and then estimates:
42 *
43 * quotient = (x * m) >> BARRETT_SHIFT
44 * remainder = x - quotient * q (corrected by reduce_once())
45 *
46 * Correctness requires BARRETT_SHIFT >= 2 * LOG2PRIME (== 2 * 12 == 24).
47 * The minimum shift of 24 bits is sufficient for the Barrett reduction
48 * bounds required by ML-KEM.
49 *
50 * On s390x, a Barrett shift of 32 allows (a * b) >> BARRETT_SHIFT to be
51 * obtained directly from the high half of a 32x32->64-bit multiplication,
52 * avoiding additional instructions.
53 * This optimization is used by the generic implementations of `reduce`
54 * and `compress`. In particular, `compress` is not manually vectorized,
55 * so it can benefit from compiler autovectorization when
56 * BARRETT_SHIFT == 32.
57 *
58 * On all other platforms BARRETT_SHIFT == 24 is used. The multiplier
59 * m = floor(2^24 / 3329) = 5039 fits in 13 bits, which the compiler can
60 * represent as a small immediate. On AArch64 the alternative value
61 * m = floor(2^32 / 3329) = 1290167 requires a wider immediate encoding
62 * (an extra movk instruction), making the 32-bit shift slightly less
63 * efficient on that platform.
64 */
65#if defined(__s390x__)
66#define BARRETT_SHIFT24 32
67#else
68#define BARRETT_SHIFT24 24
69#endif

436 * Function pointer types for NTT dispatch.
437 *
438 * Three NTT-related entry points are dispatched at runtime:
439 *
440 * scalar_ntt
441 * Forward NTT. Converts a polynomial from standard into NTT domain.
442 *
443 * scalar_inverse_ntt
444 * Full inverse NTT: butterfly stages followed by multiplication by
445 * INVERSE_DEGREE = (n/2)^-1 mod q. Used inside matrix_mult_intt where
446 * each result row is immediately transformed back to standard form.
447 *
448 * scalar_inverse_ntt_demontgomerize
449 * Inverse NTT whose input may still carry an inverse-Montgomery factor
450 * R^-1 from a preceding inner_product_montgomery call. On the generic
451 * and PPC paths the two pointers are identical (both point at the same
452 * fully-reduced implementation), because the generic inner_product
453 * already produces fully-reduced outputs via Barrett reduction. On the
454 * s390x/vec128 path the two are distinct:
455 * - inner_product_montgomery_vec128 accumulates the dot-product using
456 * Montgomery arithmetic and intentionally leaves each coefficient
457 * scaled by R^-1 (= 169 mod q), deferring the final conversion to
458 * save a per-coefficient multiply.

461 * inverse-Montgomery form to standard form, then runs the raw
462 * inverse NTT butterfly stages, and finally multiplies by
463 * INVERSE_DEGREE via scalar_mult_const_512_vec128.

466 * polynomial after the (inner_product_montgomery, scalar_inverse_ntt_demontgomerize)

622/*-
623 * reduce_once reduces 0 <= x < 2*kPrime, mod kPrime.
624 *
625 * Subtract |q| if the input is larger, without exposing a side-channel,
626 * avoiding the "clangover" attack. See |constish_time_true| for a
627 * discussion on why the value barrier is by default omitted.
628 */
629static __owur uint16_t reduce_once(uint16_t x)
630{
631 const uint16_t subtracted = x - kPrime;
632 uint16_t mask = constish_time_true(subtracted >> 15)(0u - (subtracted >> 15));
633
634 return (mask & x) | (~mask & subtracted);
635}

637/*
638 * Constant-time Barrett reduction modulo kPrime.
639 *
640 * This implementation performs a single final correction via
641 * reduce_once(), so it requires the Barrett approximation to leave
642 * the intermediate remainder below 2*kPrime.
643 *
644 * All current callers satisfy
645 *
646 * x < kPrime + 2*kPrime*kPrime,
647 *
648 * which covers products of two already reduced coefficients together
649 * with the additions performed by scalar_mult_add_generic().
650 *
651 * For BARRETT_SHIFT == 24, this bound is below the first input that
652 * would require more than one correction. For BARRETT_SHIFT == 32,
653 * a single correction is sufficient for every uint32_t input.
654 */
655static __owur uint16_t reduce(uint32_t x)
656{
657 uint64_t product = (uint64_t)x * kBarrettMultiplier;
658 uint32_t quotient = (uint32_t)(product >> kBarrettShift);
659 uint32_t remainder = x - quotient * kPrime;
660
661 return reduce_once(remainder);
662}

675/*-
676 * FIPS 203, Section 4.3, Algorithm 9: "NTT".
677 * In-place number theoretic transform of a given scalar. Note that ML-KEM's
678 * kPrime 3329 does not have a 512th root of unity, so this transform leaves
679 * off the last iteration of the usual FFT code, with the 128 relevant roots of
680 * unity being stored in NTTRoots. This means the output should be seen as 128
681 * elements in GF(3329^2), with the coefficients of the elements being
682 * consecutive entries in |s->c|.
683 */
684void ossl_ml_kem_scalar_ntt_generic(scalar *s)
685{
686 const uint16_t *roots = kNTTRoots;
687 uint16_t *end = s->c + DEGREE256;
688 int offset = DEGREE256 / 2;
689
690 do {
691 uint16_t *curr = s->c, *peer;
692
693 do {
694 uint16_t *pause = curr + offset, even, odd;
695 uint32_t zeta = *++roots;
696
697 peer = pause;
698 do {
699 even = *curr;
700 odd = reduce(*peer * zeta);
701 *peer++ = reduce_once(even - odd + kPrime);
702 *curr++ = reduce_once(odd + even);
703 } while (curr < pause);
704 } while ((curr = peer) < end);
705 } while ((offset >>= 1) >= 2);
706}

709 * FIPS 203, Section 4.3, Algorithm 10: "NTT^(-1)".
710 * In-place inverse number theoretic transform of a given scalar, with pairs of
711 * entries of s->v being interpreted as elements of GF(3329^2). Just as with
712 * the number theoretic transform, this leaves off the first step of the normal
713 * iFFT to account for the fact that 3329 does not have a 512th root of unity,
714 * using the precomputed 128 roots of unity stored in InverseNTTRoots.
715 */
716void ossl_ml_kem_scalar_inverse_ntt_generic(scalar *s)
717{
718 const uint16_t *roots = kInverseNTTRoots;
719 uint16_t *end = s->c + DEGREE256;
720 int offset = 2;
721
722 do {
723 uint16_t *curr = s->c, *peer;
724
725 do {

727 uint32_t zeta = *++roots;

733 *peer++ = reduce(zeta * (even - odd + kPrime));

735 } while (curr < pause);
736 } while ((curr = peer) < end);
737 } while ((offset <<= 1) < DEGREE256);
738 scalar_mult_const(s, kInverseDegree);
739}

760 * Multiplying two scalars in the number theoretically transformed state. Since
761 * 3329 does not have a 512th root of unity, this means we have to interpret
762 * the 2*ith and (2*i+1)th entries of the scalar as elements of
763 * GF(3329)[X]/(X^2 - 17^(2*bitreverse(i)+1)).
764 *
765 * The value of 17^(2*bitreverse(i)+1) mod 3329 is stored in the precomputed
766 * ModRoots table. Note that our Barrett transform only allows us to multiply
767 * two reduced numbers together, so we need some intermediate reduction steps,
768 * even if an uint64_t could hold 3 multiplied numbers.
769 */

771static void scalar_mult_generic(scalar *out, const scalar *lhs,
772 const scalar *rhs)
773{
774 uint16_t *curr = out->c, *end = curr + DEGREE256;
775 const uint16_t *lc = lhs->c, *rc = rhs->c;
776 const uint16_t *roots = kModRoots;
777
778 do {
779 uint32_t l0 = *lc++, r0 = *rc++;
780 uint32_t l1 = *lc++, r1 = *rc++;
781 uint32_t zetapow = *roots++;
782
783 *curr++ = reduce(l0 * r0 + reduce(l1 * r1) * zetapow);
784 *curr++ = reduce(l0 * r1 + l1 * r0);
785 } while (curr < end);
786}

970/*
971 * FIPS 203, Section 4.2.1, Equation (4.7): Compress_d.
972 *
973 * Compresses (lossily) an input |x| mod 3329 into |bits| many bits by grouping
974 * numbers close to each other together. The formula used is
975 * round(2^|bits|/kPrime*x) mod 2^|bits|.
976 * Uses Barrett reduction to achieve constant time. Since we need both the
977 * remainder (for rounding) and the quotient (as the result), we cannot use
978 * |reduce| here, but need to do the Barrett reduction directly.
979 */
980static __owur uint16_t compress(uint16_t x, int bits)
981{
982 uint32_t shifted = (uint32_t)x << bits;
983 uint64_t product = (uint64_t)shifted * kBarrettMultiplier;
984 uint32_t quotient = (uint32_t)(product >> kBarrettShift);
985 uint32_t remainder = shifted - quotient * kPrime;
986
987 /*
988 * Adjust the quotient to round correctly:
989 * 0 <= remainder <= kHalfPrime round to 0
990 * kHalfPrime < remainder <= kPrime + kHalfPrime round to 1
991 * kPrime + kHalfPrime < remainder < 2 * kPrime round to 2
992 */
993 quotient += 1 & constant_time_lt_32(kHalfPrime, remainder);
994 quotient += 1 & constant_time_lt_32(kPrime + kHalfPrime, remainder);
995 return quotient & ((1 << bits) - 1);
996}

998/*
999 * FIPS 203, Section 4.2.1, Equation (4.8): Decompress_d.
1000
1001 * Decompresses |x| by using a close equi-distant representative. The formula
1002 * is round(kPrime/2^|bits|*x). Note that 2^|bits| being the divisor allows us
1003 * to implement this logic using only bit operations.
1004 */
1005static __owur uint16_t decompress(uint16_t x, int bits)
1006{
1007 uint32_t product = (uint32_t)x * kPrime;
1008 uint32_t power = 1 << bits;
1009 /* This is |product| % power, since |power| is a power of 2. */
1010 uint32_t remainder = product & (power - 1);
1011 /* This is |product| / power, since |power| is a power of 2. */
1012 uint32_t lower = product >> bits;
1013
1014 /*
1015 * The rounding logic works since the first half of numbers mod |power|
1016 * have a 0 as first bit, and the second half has a 1 as first bit, since
1017 * |power| is a power of 2. As a 12 bit number, |remainder| is always
1018 * positive, so we will shift in 0s for a right shift.
1019 */
1020 return lower + (remainder >> (bits - 1));
1021}
